26 MAILBOX

module producer_consumer_tb;
  mailbox mbx;
  // Producer task
  task automatic producer();
    int data;
    $display("\n[Producer] Started");
    repeat (5) begin
      data = $urandom_range(1,100);
      mbx.put(data);
      $display("[Producer] Sent data: %0d", data);
      #2;
    end
    $display("[Producer] Finished\n");
  endtask

  // Consumer task
  task automatic consumer();
    int recv;
    $display("[Consumer] Started");
    repeat (5) begin
      mbx.get(recv);
      $display("[Consumer] Received data: %0d", recv);
      #3;
    end
    $display("[Consumer] Finished\n");
  endtask

  initial begin
    mbx = new();
    $display("=== PRODUCER-CONSUMER SIMULATION START ===");
    fork
      producer();
      consumer();
    join
    $display("=== SIMULATION COMPLETE ===");
    $finish;
  end
endmodule

class Rectangle;

31 SHALLOW COPY DEEP COPY
class Data;
  int arr[]; // dynamic
  function new(int n);
    arr = new[n];
    foreach (arr[i]) arr[i] = i + 1;
  endfunction
  function void display(string msg);
    $write("%s: ", msg);
    foreach (arr[i]) $write("%0d ", arr[i]);
    $display("");
  endfunction
  function Data deep_copy();
    Data temp = new(arr.size());
    foreach (arr[i]) temp.arr[i] = this.arr[i];
    return temp;
  endfunction
endclass

module copy_demo;
  Data d1, d2, d3;
  initial begin
    d1 = new(5);
    d1.display("Original d1");
    d2 = d1;           // shallow
    d2.display("After Shallow Copy d2");
    d3 = d1.deep_copy(); // deep
    d3.display("After Deep Copy d3");
    d1.arr[0] = 99;
    $display("\n--- After Modifying d1.arr[0] = 99 ---");
    d1.display("d1");
    d2.display("d2 (Shallow Copy)"); // shows changed value
    d3.display("d3 (Deep Copy)");    // unaffected
    $finish;
  end
endmodule


32 ACCOUNT CLASS LOCAL DATA METHOD
class Account;
  local real balance;
  function new(real initial_balance);
    balance = initial_balance;
  endfunction
  protected function void update_balance(real amount);
    balance = balance + amount;
  endfunction
  function void show_balance();
    $display("Current Balance: %.2f", balance);
  endfunction
endclass

class SavingsAccount extends Account;
  function new(real b);
    super.new(b);
  endfunction
  function void deposit(real amount);
    update_balance(amount); // allowed: protected method accessible in derived class
    $display("Deposited %.2f successfully.", amount);
  endfunction
endclass

module account_access_demo;
  initial begin
    SavingsAccount s1 = new(5000.0);
    s1.show_balance();
    s1.deposit(1500.0);
    s1.show_balance();
    $finish;
  end
endmodule


33 ABSTRACT CLASS TRANSACTION
virtual class Transaction;
  rand int amount;
  pure virtual function void execute();
  function void pre_randomize();
    $display("Preparing to randomize transaction...");
  endfunction
  function void post_randomize();
    $display("Transaction randomization completed. Amount = %0d", amount);
  endfunction
endclass

class Deposit extends Transaction;
  function void execute();
    if (!randomize(amount) with { amount inside {[100:10000]}; }) begin
      $display("Randomization failed for Deposit.");
    end else begin
      $display("\n[Deposit Transaction]");
      $display("Deposited Amount: %0d", amount);
    end
  endfunction
endclass

class Withdraw extends Transaction;
  function void execute();
    if (!randomize(amount) with { amount inside {[100:10000]}; }) begin
      $display("Randomization failed for Withdraw.");
    end else begin
      $display("\n[Withdraw Transaction]");
      $display("Withdrawn Amount: %0d", amount);
    end
  endfunction
endclass

module transaction_demo;
  Deposit d;
  Withdraw w;
  initial begin
    d = new();
    w = new();
    d.execute();
    w.execute();
    $finish;
  end
endmodule



34 PACKET CLASS WITH Randomization
class Packet;
  rand bit [3:0] src_addr;
  rand bit [3:0] dest_addr;
  rand bit [7:0] payload;

  function void pre_randomize();
    $display("About to randomize packet fields... Setting them to 0.");
    src_addr = 0;
    dest_addr = 0;
    payload = 0;
  endfunction

  function void post_randomize();
    $display("Randomized Packet -> src=%0d dest=%0d payload=%0d",
             src_addr, dest_addr, payload);
    if (src_addr == dest_addr)
      $display("WARNING: src_addr == dest_addr");
  endfunction
endclass

module packet_tb;
  Packet p;
  initial begin
    p = new();
    if (!p.randomize()) $display("Randomize failed");
    $finish;
  end
endmodule
