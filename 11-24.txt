// PROGRAM 11 — SEMAPHORE
// ============================================================================
module semaphore_example;
  semaphore key; // declare semaphore

  initial begin
    key = new (1); // allow only 1 key
    fork
      personA();
      personB();
      #25 personA();
    join_none
  end

  task getRoom (input bit [1:0] id);
    $display ("[%0t] Trying to get a room for id[%0d].", $time, id);
    key.get(1);
    $display ("[%0t] Room Key retrieved for id[%0d]", $time, id);
  endtask

  task putRoom (input bit [1:0] id);
    $display ("[%0t] Leaving room id[%0d].", $time, id);
    key.put(1);
    $display ("[%0t] Room Key put back id[%0d]", $time, id);
  endtask

  task personA ();
    getRoom(1);
    #20 putRoom(1);
  endtask

  task personB ();
    #5 getRoom(2);
    #10 putRoom(2);
  endtask
endmodule

// PROGRAM 13 — DYNAMIC ARRAY METHODS
// ============================================================================
module dynamic_array_methods;
  int dyn_arr[];
  int insert_pos;
  int new_val;
  int del_pos;

  initial begin
    dyn_arr = new[3];
    dyn_arr = '{10, 20, 30};
    $display("Initial Array: %p, size=%0d", dyn_arr, dyn_arr.size());

    // Insert
    insert_pos = 1;
    new_val = 15;
    dyn_arr = new[dyn_arr.size()+1](dyn_arr);
    for (int i = dyn_arr.size()-1; i > insert_pos; i--)
      dyn_arr[i] = dyn_arr[i-1];
    dyn_arr[insert_pos] = new_val;
    $display("After Insert: %p", dyn_arr);

    // Delete
    del_pos = 2;
    for (int i = del_pos; i < dyn_arr.size()-1; i++)
      dyn_arr[i] = dyn_arr[i+1];
    dyn_arr = new[dyn_arr.size()-1](dyn_arr);
    $display("After Delete: %p", dyn_arr);

    dyn_arr.delete();
    $display("After Deleting Entire Array: size=%0d", dyn_arr.size());
  end
endmodule

// PROGRAM 14 — ASSOCIATIVE ARRAY METHODS
// ============================================================================
module associative_array_example;
  int assoc_arr[int];
  int key;

  initial begin
    assoc_arr[10] = 100;
    assoc_arr[5]  = 50;
    assoc_arr[20] = 200;

    if (assoc_arr.exists(10))
      $display("Key 10 exists with value = %0d", assoc_arr[10]);

    if (assoc_arr.first(key))
      $display("First key = %0d, value = %0d", key, assoc_arr[key]);

    if (assoc_arr.last(key))
      $display("Last key = %0d, value = %0d", key, assoc_arr[key]);

    if (assoc_arr.first(key)) begin
      do begin
        $display("Ascending: key = %0d, value = %0d", key, assoc_arr[key]);
      end while (assoc_arr.next(key));
    end

    if (assoc_arr.last(key)) begin
      do begin
        $display("Descending: key = %0d, value = %0d", key, assoc_arr[key]);
      end while (assoc_arr.prev(key));
    end
  end
endmodule

// PROGRAM 16 — VENDING MACHINE (IF & CASE)
// ============================================================================
module vending_machine;
  int choice;

  initial begin
    choice = 2;

    if (choice == 1) $display("Dispensing: Soda");
    else if (choice == 2) $display("Dispensing: Chips");
    else if (choice == 3) $display("Dispensing: Chocolate");
    else $display("Invalid choice");

    case(choice)
      1: $display("Case: Soda");
      2: $display("Case: Chips");
      3: $display("Case: Chocolate");
      default: $display("Case: Invalid choice");
    endcase
  end
endmodule

// PROGRAM 17 — TASK VS FUNCTION
// ============================================================================
module task_vs_function;

  function int square(input int a);
    square = a * a;
  endfunction

  task multiply(input int a, input int b, output int res);
    res = a * b;
  endtask

  initial begin
    int sq, mul;
    sq = square(5);
    $display("Square(5) = %0d", sq);

    multiply(3, 7, mul);
    $display("Multiply(3,7) = %0d", mul);
  end
endmodule
